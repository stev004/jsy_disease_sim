import sys
import tempfile
from pathlib import Path

root = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(root / "src"))
from jersey_outbreak import g35b_design_study as g  # noqa: E402
from jersey_outbreak import phase0_campaign as p0  # noqa: E402
from jersey_outbreak.observation import observe_latent_run  # noqa: E402
from jersey_outbreak.outbreak_runner import load_parameter_set, run_outbreak  # noqa: E402
from jersey_outbreak.parent_build import build_parent  # noqa: E402

SEED, OBS = 123, 123  # CI demo seed: not a study, probe or earlier-campaign seed
params = load_parameter_set(root)
truth = p0.CandidateCell(0.08, 2, 0.75, 0.25)
with tempfile.TemporaryDirectory() as tmp:
    generated = build_parent(root, "ci", SEED, Path(tmp) / "p", write_m4=True).generated
    out = {}
    for label, cfg in (("30", g.StudyConfig()), ("49", g._ArmIIConstructorConfig())):
        rc = p0._run_config_for_cell(cfg, params, SEED, truth)
        latent = run_outbreak(generated, rc, params)
        obs = p0._observation_config_for_cell(root, cfg, OBS, "smoke-not-study", truth)
        observed = observe_latent_run(latent, obs)
        table = p0.observed_tables_from_events(
            observed.observation_events, dates=p0.observation_dates(cfg.start_date, cfg.duration_days, 4)
        )
        out[label] = (latent, table)
    res = g.prefix_identity(out["30"][0].transmission_events, out["49"][0].transmission_events,
                            out["30"][1], out["49"][1], start_date=g.StudyConfig().start_date)
    print("RESULT", res["result"], res["label"])
    if res["result"] != "identical":
        last = res["prefix_last_date"]
        a = [e for e in out["30"][0].transmission_events if str(e["infection_date"]) <= last]
        b = [e for e in out["49"][0].transmission_events if str(e["infection_date"]) <= last]
        print("counts", len(a), len(b))
        ka = {k for e in a for k in e}
        diff = sorted(k for k in ka if sorted(str(e.get(k)) for e in a) != sorted(str(e.get(k)) for e in b))
        print("fields that differ:", diff)
    print("manip", g.manipulation_check(out["49"][0].daily_epidemic))
