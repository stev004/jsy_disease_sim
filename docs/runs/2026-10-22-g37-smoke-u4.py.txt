import sys
import tempfile
from datetime import timedelta
from fractions import Fraction
from pathlib import Path

root = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(root / "src"))
from jersey_outbreak import g35b_design_study as g  # noqa: E402
from jersey_outbreak import phase0_campaign as p0  # noqa: E402
from jersey_outbreak.outbreak_runner import load_parameter_set, run_outbreak  # noqa: E402
from jersey_outbreak.parent_build import build_parent  # noqa: E402

SEED = 123  # CI demo seed: not a study, probe or earlier-campaign seed
params = load_parameter_set(root)
truth = p0.CandidateCell(0.08, 2, 0.75, 0.25)
cfg = g.StudyConfig()
with tempfile.TemporaryDirectory() as tmp:
    generated = build_parent(root, "ci", SEED, Path(tmp) / "p", write_m4=True).generated
    latent = run_outbreak(generated, p0._run_config_for_cell(cfg, params, SEED, truth), params)
    ev = latent.transmission_events[0]
    print("infection_date type:", type(ev["infection_date"]).__name__, repr(ev["infection_date"]))
    d = cfg.start_date + timedelta(days=29)
    N = len(generated.agent_ids)
    census = g.observe_serosurvey(latent, process_seed=SEED, serosurvey_seed=1, config_id="smoke-not-study", n=N, survey_date=d)
    row = [r for r in latent.daily_epidemic if r["date"] == d.isoformat()][0]
    print("census k", census.k, "N", N, "fraction*N", row["ever_infected_fraction"] * N,
          "exact match:", Fraction(census.k, N) == Fraction(row["ever_infected_fraction"]).limit_denominator(N))
    s = g.observe_serosurvey(latent, process_seed=SEED, serosurvey_seed=1, config_id="smoke-not-study", n=400, survey_date=d)
    print("n=400 k", s.k, "z", g.anscombe_z(s.k, 400))
    episodes = len([e for e in latent.transmission_events if str(e["infection_date"]) <= d.isoformat()])
    print("episodes<=d", episodes, "(>= census k)")
