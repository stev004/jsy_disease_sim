print("\n#### 5. serosurvey sampling order (SHA-256 rule) and frame")
import pyarrow.parquet as pq
ss = J("arm_iii/arm_iii/serosurvey_samples.json")
print("design:", ss["design"])
def frame(seed):
    d = glob.glob(f"*/parents/{seed}/populations/*/residents.parquet")[0]
    t = pq.read_table(d)
    col = "agent_id" if "agent_id" in t.column_names else t.column_names[0]
    return [str(x) for x in t.column(col).to_pylist()], col
order_bad = []
for rec in ss["targets"] + ss["candidates"]:
    ids, col = frame(rec["process_seed"])
    key = lambda a: hashlib.sha256(f"{rec['process_seed']}|{rec['serosurvey_seed']}|{rec['config_id']}|serosurvey|{a}".encode()).hexdigest()
    exp = sorted(ids, key=key)[:400]
    if exp != rec["sampled_agent_ids"] or len(ids) != 3000 or rec["survey_date"] != "2025-02-04":
        order_bad.append((rec["process_seed"], rec.get("beta"), rec.get("offset"), len(ids), rec["survey_date"]))
print("frame column:", col, " records:", len(ss["targets"]) + len(ss["candidates"]), " order/frame/date violations:", order_bad)
