def to_dec(name, value):
    """Map a float grid value to its declared decimal string (exact)."""
    for d in DIMS:
        if d[0] == name:
            matches = [s for s in d[2] if float(s) == float(value)]
            if len(matches) != 1:
                raise SystemExit(f"value {value!r} not on declared grid for {name}")
            return Decimal(matches[0])
    raise KeyError(name)


def cell_key(cell):
    return tuple(float(cell[n]) for n in DIM_NAMES)


def load_json(path):
    with open(path) as fh:
        return json.load(fh)


def load_arm_tables(arm_dir):
    targets = {}
    for s in TARGETS:
        targets[s] = load_json(os.path.join(arm_dir, "target_tables", f"{s}.json"))
    lib = {}  # cell_key -> {seed: record}
    cdir = os.path.join(arm_dir, "candidate_tables")
    for idx in os.listdir(cdir):
        for fn in os.listdir(os.path.join(cdir, idx)):
            rec = load_json(os.path.join(cdir, idx, fn))
            k = cell_key(rec["cell"])
            seed = int(fn.split(".")[0])
            if seed != rec["process_seed"]:
                raise SystemExit(f"file/seed mismatch {idx}/{fn}")
            lib.setdefault(k, {})
            if seed in lib[k]:
                raise SystemExit(f"duplicate replicate {k} {seed}")
            lib[k][seed] = rec
    return targets, lib


def channels(table):
    return [table["symptomatic"], table["asymptomatic"]]


def rep_sum(target, preds, T):
    """Σ_c Σ_t (sqrt(y+3/8) - mean_j sqrt(x_j+3/8))^2 in Phase-0 order (sym, asym)."""
    tch = channels(target)
    pch = [channels(p) for p in preds]
    R = len(preds)
    total = 0.0
    for c in range(2):
        if len(tch[c]) != T:
            raise SystemExit("target T mismatch")
        for t in range(T):
            m = sum(math.sqrt(p[c][t] + 3.0 / 8.0) for p in pch) / R
            total += (math.sqrt(tch[c][t] + 3.0 / 8.0) - m) ** 2
    return total


def anscombe_z(k, n):
    return math.sqrt(n + 0.5) * math.asin(math.sqrt((k + 3.0 / 8.0) / (n + 3.0 / 4.0)))


def fit(surface):
    """surface: dict cell_key -> objective. Frozen §6 rules."""
    if len(surface) != 625:
        raise SystemExit("surface not 625")
    L_min = min(surface.values())
    tau = 1e-12 * max(1.0, L_min)
    minimizers = [k for k, v in surface.items() if abs(v - L_min) <= tau]
    tied = len(minimizers) > 1
    selected = None if tied else minimizers[0]
    profiles = {}
    for i, (name, truth, grid, tol, bias) in enumerate(DIMS):
        vals = [float(g) for g in grid]
        P = [min(v for k, v in surface.items() if k[i] == gv) for gv in vals]
        Pmin = min(P)
        n_at_min = sum(1 for p in P if abs(p - Pmin) <= tau)
        above = sorted(p for p in P if p > Pmin + tau)
        second = above[0] if above else None
        gap = None if second is None else (second - Pmin) / max(Pmin, 1e-12)
        ident = n_at_min == 1 and gap is not None and gap >= 0.05
        profiles[name] = {
            "P": P, "Pmin": Pmin, "argmin": vals[P.index(Pmin)], "unique": n_at_min == 1,
            "second": second, "gap": gap, "identified": ident,
        }
    return {"L_min": L_min, "tau": tau, "tied": tied, "selected": selected,
            "minimizers": minimizers, "profiles": profiles}


def predicates(fits):
    """§7.1 predicates 4-8 with exact integer and decimal arithmetic."""
    hits = {n: 0 for n in DIM_NAMES}
    boundary = {n: 0 for n in DIM_NAMES}
    errs = {n: [] for n in DIM_NAMES}
    joint = 0
    per_target = {}
    for s in TARGETS:
        f = fits[s]
        all_hit = True
        row = {}
        for i, (name, truth, grid, tol, bias) in enumerate(DIMS):
            sel = f["selected"]
            if sel is None:
                err = None
                within = False
                atb = False
            else:
                est = to_dec(name, sel[i])
                err = est - Decimal(truth)
                within = abs(err) <= Decimal(tol)
                atb = est in (Decimal(grid[0]), Decimal(grid[-1]))
                errs[name].append(err)
            ident = (not f["tied"]) and f["profiles"][name]["identified"]
            hit = ident and within
            hits[name] += int(hit)
            boundary[name] += int(atb)
            all_hit = all_hit and hit
            row[name] = {"est": None if sel is None else str(to_dec(name, sel[i])),
                         "err": None if err is None else str(err),
                         "identified": ident, "within": within, "hit": hit,
                         "boundary": atb, "gap": f["profiles"][name]["gap"]}
        joint += int(all_hit)
        row["joint"] = all_hit
        row["tied"] = f["tied"]
        per_target[s] = row
    mean_bias = {n: (sum(errs[n], Decimal(0)) / Decimal(10) if len(errs[n]) == 10 else None) for n in DIM_NAMES}
    p = {
        "4_no_global_tie": all(not fits[s]["tied"] for s in TARGETS),
        "5_dimension_hits": all(hits[n] * 5 >= 4 * 10 for n in DIM_NAMES),
        "6_joint_hits": joint * 5 >= 3 * 10,
        "7_bias": all(mean_bias[n] is not None and abs(mean_bias[n]) <= Decimal(d[4]) for n, d in zip(DIM_NAMES, DIMS)),
        "8_boundaries": all(boundary[n] * 5 <= 1 * 10 for n in DIM_NAMES),
    }
    return {"hits": hits, "boundary": boundary, "joint": joint,
            "mean_bias": {n: (None if v is None else str(v)) for n, v in mean_bias.items()},
            "predicates": p, "per_target": per_target}


def compare_surface(arm_dir, sub, s, surface):
    """Compare recomputed surface with the stored blind estimate surface."""
    path = os.path.join(arm_dir, sub, f"{s}.json")
    blind = load_json(path)
    stored = {cell_key(r): r["objective"] for r in blind["loss_surface"]}
    if set(stored) != set(surface):
        raise SystemExit(f"cell set mismatch {path}")
    maxdiff = max(abs(stored[k] - surface[k]) for k in surface)
    exact = sum(1 for k in surface if stored[k] == surface[k])
    stored_sel = None if blind["selected"] is None else cell_key(blind["selected"])
    return blind, stored, maxdiff, exact, stored_sel


def truth_viability(target_table):
    sym = sum(target_table["symptomatic"])
    asym = sum(target_table["asymptomatic"])
    nz = sum(1 for a, b in zip(target_table["symptomatic"], target_table["asymptomatic"]) if a + b > 0)
    return {"sym_reports": sym, "asym_reports": asym, "nonzero_dates": nz,
            "table_part_ok": sym > 0 and asym > 0 and nz >= 3}


def run_arm(label, arm_dir, blind_sub, T, R_seeds, sero=None):
    targets, lib = load_arm_tables(arm_dir)
    # completeness
    complete_cells = len(lib) == 625
    expected_seeds = sorted(R_seeds)
    reps_ok = all(sorted(v) == expected_seeds for v in lib.values())
    calendars_ok = all(len(t["dates"]) == T for t in targets.values()) and all(
        r["table"]["dates"] == targets[TARGETS[0]]["dates"] for v in lib.values() for r in v.values())
    fits, comps = {}, {}
    sero_terms = {}
    for s in TARGETS:
        surface = {}
        for k, reps in lib.items():
            preds = [reps[seed]["table"] for seed in R_seeds]
            total = rep_sum(targets[s], preds, T)
            if sero is not None:
                zt = anscombe_z(sero["target_k"][s], SERO_N)
                zbar = sum(anscombe_z(sero["cand_k"][(seed, k[0], k[1])], SERO_N) for seed in R_seeds) / len(R_seeds)
                total = total + (zt - zbar) ** 2
            surface[k] = total / (2.0 * T)
        f = fit(surface)
        fits[s] = f
        blind, stored, maxdiff, exact, stored_sel = compare_surface(arm_dir, blind_sub, s, surface)
        # also compute fit on the stored surface (as-claimed) to see whether any decision differs
        f_stored = fit(stored)
        comps[s] = {"max_abs_objective_diff": maxdiff, "bit_exact_cells": exact,
                    "selected_recomputed": f["selected"], "selected_stored": stored_sel,
                    "selection_agrees": f["selected"] == stored_sel,
                    "identified_flags_agree": all(
                        f["profiles"][n]["identified"] == next(p["identified"] for p in blind["profiles"] if p["dimension"] == n)
                        for n in DIM_NAMES),
                    "stored_surface_fit_selected": f_stored["selected"],
                    "stored_minimum": blind["minimum_objective"], "recomputed_minimum": f["L_min"],
                    "estimate_hash": blind["estimate_hash"]}
    pred = predicates(fits)
    tv = {s: truth_viability(targets[s]) for s in TARGETS}
    return {"label": label, "complete_cells": complete_cells, "replicates_ok": reps_ok,
            "calendars_ok": calendars_ok, "T": T, "R": len(R_seeds),
            "fits": {s: {"selected": f["selected"], "tied": f["tied"], "L_min": f["L_min"],
                         "profiles": f["profiles"]} for s, f in fits.items()},
            "comparison": comps, "pred": pred, "truth_tables": tv, "lib": lib, "targets": targets}


def load_claims(arm_dir, code):
    ev = load_json(os.path.join(arm_dir, "recovery_evaluation.json"))
    rows = list(csv.DictReader(open(os.path.join(arm_dir, f"g35b_{code}_recovery.csv"))))
    return ev, rows
